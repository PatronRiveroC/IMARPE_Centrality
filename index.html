Inferencias sobre los registros

library(ggplot2)
library(ggpubr)
library(dismo)

inf_vals <- raster::extract(ellipsoidMod$suitRaster, data[, c(1:2)])
if (is.matrix(inf_vals) || is.data.frame(inf_vals)) {
  inf_vals <- inf_vals[, ncol(inf_vals)]
}
inf_dnc <- 1 - inf_vals

formula_lm <- abun ~ poly(DNC, 2, raw = TRUE)
ajuste <- lm(formula_lm, data = cor_data)
abun_esperada <- predict(ajuste, newdata = data.frame(DNC = inf_dnc))

df_esperados <- data.frame(DNC = inf_dnc, abun = abun_esperada)

ggplot(cor_data, aes(DNC, abun)) +
  geom_point(shape = 1, size = 1.5, color = "black") +
  geom_point(data = df_esperados, aes(DNC, abun), shape = 4, size = 2, color = "blue", stroke = 1) +
  stat_cor(method = "spearman", cor.coef.name = "rho", label.x.npc = "left", label.y.npc = "top") +
  labs(x = "DNC", y = "Abundance") +
  geom_smooth(color = "red", method = "lm", formula = y ~ poly(x, 2, raw = TRUE), se = TRUE) +
